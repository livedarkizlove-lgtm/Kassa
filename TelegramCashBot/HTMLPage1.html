# -*- coding: utf-8 -*- from flask import Flask, render_template, request, redirect from tinydb import TinyDB, Query app = Flask(__name__) db_products = TinyDB('../data/products.json') db_history = TinyDB('../data/history.json') # Главная страница @app.route('/') def index(): products = db_products.all() history = db_history.all() return render_template('index.html', products=products, history=history) # Добавление товара @app.route('/add', methods=['POST']) def add(): name = request.form['name'] price = float(request.form['price']) db_products.insert({"name": name, "price": price, "stock": 0}) db_history.insert({"action": "Добавление товара", "product": name, "amount": price}) return redirect('/') # Удаление товара @app.route('/delete', methods=['POST']) def delete(): name = request.form['name'] Product = Query() db_products.remove(Product.name == name) db_history.insert({"action": "Удаление товара", "product": name}) return redirect('/') # Пополнение/Списание @app.route('/update', methods=['POST']) def update(): name = request.form['name'] amount = int(request.form['amount']) Product = Query() item = db_products.search(Product.name == name) if item: new_stock = item[0]['stock'] + amount db_products.update({'stock': new_stock}, Product.name == name) action = "Пополнение" if amount > 0 else "Списание" db_history.insert({"action": action, "product": name, "amount": amount}) return redirect('/') # Запуск сервера if __name__ == '__main__': app.run(debug=True)
